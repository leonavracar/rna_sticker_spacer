Calibrating chain stiffness to persistence length¶

*consider PBC

  1. One polyA chain l = 200
  2. Pick values for ppp_angle_k to sweep over + many steps + saves
  3. Calulate tangent/bond-vector correlation along the chain:
    • bonds.shape == (199,3) for 200 beads
    • bond vector, normalize to unit vector (direction), t_i = b_i / |b_i|
    • compare bond i to bond i+s (compute all dot products t_i · t_{i+s})
    • average over all valid i
    • average over all saved snapshots
    • repeat over all s
    • Plot: x (monomer separation), y (<cos(phi)>)

import re
from pathlib import Path

import gsd.hoomd
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit


BASE_DIR = Path(
    "/groups/goloborodko/projects/leona/validation/bulk/output_dir/"
    "46a_1k_ppp_angle_persistence_calibration"
)

MAX_SEP = 50          # change
REPLICATES = [0, 1, 2,3,4]
USE_LAST_N_FRAMES = None   # 100 if you only want last 100 frames


def get_bond_vectors(coords):
    """
    coords shape: (N, 3)

    Returns bond vectors:
        b_i = r_{i+1} - r_i
    """
    return coords[1:] - coords[:-1]


def get_avg_cos_theta_from_coords(coords, max_sep=50):
    """
    Computes <cos(theta_s)> for separations s = 1 ... max_sep.

    Formula:
        <cos(theta_s)> = mean_i [ b_hat_i dot b_hat_{i+s} ]
    """
    bond_vectors = get_bond_vectors(coords)

    norms = np.linalg.norm(bond_vectors, axis=1, keepdims=True)

    # Avoid division by zero if something weird happens
    valid = norms[:, 0] > 0
    bond_vectors = bond_vectors[valid]
    norms = norms[valid]

    unit_vectors = bond_vectors / norms

    n_bonds = len(unit_vectors)
    max_sep = min(max_sep, n_bonds - 1)

    avg_cos = []

    for s in range(1, max_sep + 1):
        dots = np.sum(unit_vectors[:-s] * unit_vectors[s:], axis=1)
        avg_cos.append(np.mean(dots))

    return np.array(avg_cos)


def unwrap_positions(frame):
    """
    Uses GSD particle images to unwrap coordinates if available.

    This matters if the chain crosses periodic boundaries.
    """
    pos = frame.particles.position.copy()

    if hasattr(frame.particles, "image") and frame.particles.image is not None:
        image = frame.particles.image
        box = frame.configuration.box
        lx, ly, lz = box[0], box[1], box[2]

        pos[:, 0] += image[:, 0] * lx
        pos[:, 1] += image[:, 1] * ly
        pos[:, 2] += image[:, 2] * lz

    return pos


def avg_cos_from_gsd(gsd_path, max_sep=50, use_last_n_frames=None):
    """
    Reads a trajectory and averages <cos(theta_s)> across frames.
    """
    curves = []

    with gsd.hoomd.open(str(gsd_path), mode="r") as traj:
        if use_last_n_frames is None:
            frames = range(len(traj))
        else:
            start = max(0, len(traj) - use_last_n_frames)
            frames = range(start, len(traj))

        for frame_idx in frames:
            frame = traj[frame_idx]
            coords = unwrap_positions(frame)

            curve = get_avg_cos_theta_from_coords(coords, max_sep=max_sep)
            curves.append(curve)

    return np.mean(curves, axis=0)


def extract_angle_k(folder_name):
    """
    Extracts angle stiffness from folder names like:
        7_molecule_file_a200.json_E_att_0_angle_k_ppp_200_structure_type_...
    """
    match = re.search(r"angle_k_ppp_([0-9.]+)", folder_name)

    if match is None:
        return None

    value = match.group(1)

    if "." in value:
        return float(value)

    return int(value)


def fit_persistence_length(separations, avg_cos):
    """
    Fits:
        <cos(theta_s)> = exp(-s / lp)

    This assumes average bond length is roughly 1.
    """
    separations = np.asarray(separations)
    avg_cos = np.asarray(avg_cos)

    # Exponential fit only makes sense for positive correlation values
    mask = np.isfinite(avg_cos) & (avg_cos > 0)

    x = separations[mask]
    y = avg_cos[mask]

    if len(x) < 5:
        return np.nan

    def model(s, lp):
        return np.exp(-s / lp)

    try:
        params, cov = curve_fit(
            model,
            x,
            y,
            p0=[10.0],
            bounds=(0.001, np.inf),
            maxfev=10000,
        )
        return params[0]

    except RuntimeError:
        return np.nan

            max_sep=MAX_SEP,
            use_last_n_frames=USE_LAST_N_FRAMES,
        )

        rep_curves.append(curve)

    if len(rep_curves) > 0:
        results[angle_k] = np.array(rep_curves)

# Estimate and plot persistence length
# ------------------------------------------------------------

lp_values = {}

for angle_k in sorted(results):
    rep_curves = results[angle_k]

    lp_reps = []

    for curve in rep_curves:
        lp = fit_persistence_length(separations, curve)
        lp_reps.append(lp)

    lp_values[angle_k] = np.array(lp_reps)


angle_ks = np.array(sorted(lp_values))
lp_means = np.array([np.nanmean(lp_values[k]) for k in angle_ks])
lp_sems = np.array([
    np.nanstd(lp_values[k], ddof=1) /
    np.sqrt(np.sum(np.isfinite(lp_values[k])))
    for k in angle_ks
])

plt.figure(figsize=(8, 5))

plt.errorbar(
    angle_ks,
    lp_means,
    yerr=lp_sems,
    marker="o",
    capsize=4,
    linestyle="-",
)

plt.xlabel("angle stiffness")
plt.ylabel(r"$l_p$")
plt.title(r"Persistence length estimate")
