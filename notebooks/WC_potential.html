import numpy as np
import matplotlib.pyplot as plt
import matplotlib.gridspec as gridspec

# Parameters from simulation
r_r = 0.25
r_a = 0.75
E_r = 20.0
E_a = -1.0  # free parameter, shown as -1 for illustration
r_e = (r_r + r_a) / 2
r_l = 0.48
r_u = 0.807
n_cut = 1.061

r = np.linspace(0, 1.5, 2000)
n = np.linspace(0, 3, 1000)

# --- Functions ---
def V_r(r):
    out = np.zeros_like(r)
    mask = (r > 0) & (r < r_r)
    out[mask] = E_r * (1 - (r[mask]/r_r)**2)**2
    return out

def V_a(r):
    out = np.zeros_like(r)
    mask = (r > r_r) & (r < r_a)
    out[mask] = E_a * (1 - ((r[mask] - r_e)/((r_a - r_r)/2))**2)**2
    return out

def phi(r):
    out = np.zeros_like(r)
    out[r < r_l] = 1.0
    mask = (r >= r_l) & (r <= r_u)
    out[mask] = (1 - ((r[mask] - r_l)/(r_u - r_l))**2)**2
    return out

def omega(n, ncut):
    out = np.zeros_like(n)
    out[n < ncut] = 1.0
    mask = (n >= ncut) & (n <= ncut + 1)
    out[mask] = (1 - (n[mask] - ncut)**2)**2
    return out

# --- Figure with 3 panels ---
fig = plt.figure(figsize=(14, 4.5))
gs = gridspec.GridSpec(1, 3, wspace=0.38)

# Panel 1: phi(r)
ax1 = fig.add_subplot(gs[0])
ax1.plot(r, phi(r), color='#c0392b', lw=2)
ax1.axvline(r_l, color='gray', ls='--', lw=1)
ax1.axvline(r_u, color='gray', ls='--', lw=1)
ax1.text(r_l, 1.06, r'$r_\ell$', ha='center', fontsize=11)
ax1.text(r_u, 1.06, r'$r_u$', ha='center', fontsize=11)
ax1.set_xlabel(r'$r$', fontsize=12)
ax1.set_ylabel(r'$\phi(r)$', fontsize=12)
ax1.set_title(r'Density kernel $\phi(r)$', fontsize=12)
ax1.set_xlim(0, 1.5)
ax1.set_ylim(-0.1, 1.2)
ax1.axhline(0, color='black', lw=0.8)

# Panel 2: Omega(n)
ax2 = fig.add_subplot(gs[1])
ax2.plot(n, omega(n, n_cut), color='black', lw=2)
ax2.axvline(n_cut, color='gray', ls='--', lw=1)
ax2.axvline(n_cut+1, color='gray', ls='--', lw=1)
ax2.text(n_cut, 1.06, r'$n_\mathrm{cut}$', ha='center', fontsize=11)
ax2.text(n_cut+1, 1.06, r'$n_\mathrm{cut}+1$', ha='center', fontsize=11)
ax2.set_xlabel(r'$n$  (neighbor count)', fontsize=12)
ax2.set_ylabel(r'$\Omega(n)$', fontsize=12)
ax2.set_title(r'Switching function $\Omega(n)$', fontsize=12)
ax2.set_xlim(0, 3)
ax2.set_ylim(-0.1, 1.2)
ax2.axhline(0, color='black', lw=0.8)

# Panel 3: Full potential
ax3 = fig.add_subplot(gs[2])
n_i_vals = [0.0, 1.5, 2.5]
colors = ['#2980b9', '#8e44ad', '#e67e22']
labels = [r'$n_i=n_j=0$ (free)', r'$n_i=n_j=1.5$ (partial)', r'$n_i=n_j=2.5$ (saturated)']

for ni, col, lab in zip(n_i_vals, colors, labels):
    psi = omega(np.array([ni]), n_cut)[0]**2
    V = V_r(r) + psi * V_a(r)
    ax3.plot(r, V, color=col, lw=2, label=lab)

ax3.plot(r, V_a(r), color='gray', lw=1.5, ls='--', label=r'$U_a(r)$')
ax3.axvline(r_r, color='gray', ls=':', lw=1)
ax3.axvline(r_a, color='gray', ls=':', lw=1)
ax3.axvline(r_e, color='gray', ls=':', lw=1)
ax3.text(r_r, 0.12, r'$r_r$', ha='center', fontsize=10, color='gray')
ax3.text(r_a, 0.12, r'$r_a$', ha='center', fontsize=10, color='gray')
ax3.text(r_e, 0.12, r'$r_e$', ha='center', fontsize=10, color='gray')
ax3.axhline(0, color='black', lw=0.8)
ax3.set_xlabel(r'$r$', fontsize=12)
ax3.set_ylabel(r'$U(r)$', fontsize=12)
ax3.set_title(r'Full potential $U(r)$', fontsize=12)
ax3.set_xlim(0, 1.2)
ax3.set_ylim(-1.3, 1.5)
ax3.legend(fontsize=8, loc='upper right')

fig.text(0.5, -0.04,
         r'Note: $E_{rep} = 20$, repulsion truncated for clarity. $E_{att} = 1$ shown as illustrative value.',
         ha='center', fontsize=9, color='gray')

plt.suptitle('Watson-Crick Soft-Core Potential', fontsize=13, fontweight='bold', y=1.02)
plt.savefig('wc_potential_panels.png', dpi=150, bbox_inches='tight')
plt.show()
