ViennaRNA and secondary structure¶

  1. Recall, precision

MFE recall = fraction of Vienna MFE pairs present precision = fraction of simulated pairs that are Vienna MFE pairs F1 score = combined precision/recall base-pair distance = missing MFE pairs + extra simulated pairs

Parameter A: high recall, high precision → forms the Vienna hairpin correctly

Parameter B: high recall, low precision → forms the Vienna stem but also creates extra/wrong pairs

Parameter C: low recall, high precision → rarely forms many pairs, but the few it forms are usually correct

Parameter D: low recall, low precision → bad match to ViennaRNA

  1. Base pair occupancy
  2. Ensemble diversity (average base-pair distance between pairs of simulation frames) (optional)

def plot_mfe_metric_over_time(
    df: pd.DataFrame,
    ax: plt.Axes,
    y_col: str = "mfe_f1_score",
    x_col: str = "step",
    group_col: str = "angle_k_ppp",
    style: PlotStyle = PlotStyle(),
    run_col: str = "run_id",
    x_scale: float = 1e6,
    band_mode: str = "replicates",
    smoothing_window: int | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
) -> plt.Axes:
    """
    Plot a framewise MFE agreement metric over time.

    One mean line is drawn per group_col value, with replicate trajectories
    or replicate spread shown according to band_mode.
    """
    required_cols = [x_col, y_col, group_col, run_col]
    missing_cols = [col for col in required_cols if col not in df.columns]

    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    plot_df = df.dropna(subset=required_cols).copy()

    if x_scale != 1:
        x_plot_col = f"{x_col}_million"
        plot_df[x_plot_col] = plot_df[x_col] / x_scale
    else:
        x_plot_col = x_col

    plot_lines_by_group(
        df=plot_df,
        ax=ax,
        x_col=x_plot_col,
        y_col=y_col,
        group_col=group_col,
        style=style,
        run_col=run_col,
        label_map=label_map,
        band_mode=band_mode,
        smoothing_window=smoothing_window,
    )

    if y_col in {"mfe_recall", "mfe_precision", "mfe_f1_score"}:
        ax.set_ylim(0, 1.05)

    if x_scale == 1e6 and x_col == "step":
        ax.set_xlabel("Simulation step (×10⁶)")

    return ax

def plot_mfe_precision_recall_quadrants(
    df: pd.DataFrame,
    ax: plt.Axes,
    color_col: str | None = None,
    run_col: str = "run_id",
    recall_col: str = "mfe_recall",
    precision_col: str = "mfe_precision",
    recall_threshold: float = 0.8,
    precision_threshold: float = 0.8,
    style: PlotStyle | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
    show_quadrant_labels: bool = True,
    show_legend: bool = False,
) -> plt.Axes:
    """
    Plot replicate-level MFE precision vs recall.

    Each point is one replicate summarized over the rows passed to the function.
    If plot_panels_by_group(final_fraction=0.2) is used, each point is the
    final-window replicate mean.

    Quadrants:
    high recall + high precision: complete and specific
    high recall + low precision: complete but noisy
    low recall + high precision: partial but specific
    low recall + low precision: missing and noisy
    """
    if style is None:
        style = PlotStyle()

    required_cols = [run_col, recall_col, precision_col]
    if color_col is not None:
        required_cols.append(color_col)

    missing_cols = [col for col in required_cols if col not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    group_cols = [run_col]
    if color_col is not None:
        group_cols.append(color_col)

    rep_df = (
        df.dropna(subset=required_cols)
        .groupby(group_cols, as_index=False)
        .agg(
            recall_mean=(recall_col, "mean"),
            precision_mean=(precision_col, "mean"),
        )
    )

    ax.axvline(recall_threshold, linestyle="--", linewidth=1, alpha=0.7)
    ax.axhline(precision_threshold, linestyle="--", linewidth=1, alpha=0.7)

    if color_col is None:
        ax.scatter(
            rep_df["recall_mean"],
            rep_df["precision_mean"],
            s=style.point_size * 4,
            alpha=style.alpha,
            edgecolors="none",
        )
    else:
        default_cycle = plt.rcParams["axes.prop_cycle"].by_key()["color"]
        color_values = sorted(rep_df[color_col].dropna().unique())

        for idx, value in enumerate(color_values):
            d = rep_df[rep_df[color_col] == value]
            color = default_cycle[idx % len(default_cycle)]

            ax.scatter(
                d["recall_mean"],
                d["precision_mean"],
                s=style.point_size * 10,
                alpha=0.65,
                edgecolors="black",
                color=color,
                label=f"{get_label(color_col, label_map)}={value}, n={len(d)}",
            )

    if show_quadrant_labels:
        ax.text(
            0.97, 0.97,
            "complete\n+ specific",
            transform=ax.transAxes,
            ha="right",
            va="top",
            fontsize=style.tick_size,
        )
        ax.text(
            0.97, 0.05,
            "complete\nbut noisy",
            transform=ax.transAxes,
            ha="right",
            va="bottom",
            fontsize=style.tick_size,
        )
        ax.text(
            0.05, 0.97,
            "partial\nbut specific",
            transform=ax.transAxes,
            ha="left",
            va="top",
            fontsize=style.tick_size,
        )
        ax.text(
            0.05, 0.05,
            "missing\n+ noisy",
            transform=ax.transAxes,
            ha="left",
            va="bottom",
            fontsize=style.tick_size,
        )

    ax.set_xlim(-0.02, 1.02)
    ax.set_ylim(-0.02, 1.02)

    ax.set_xlabel(get_label(recall_col, label_map))
    ax.set_ylabel(get_label(precision_col, label_map))

    if style.show_grid:
        ax.grid(True, alpha=style.grid_alpha)

    if show_legend and color_col is not None:
        ax.legend(loc=style.legend_loc)

    return ax

def add_global_group_legend(
    fig: plt.Figure,
    df: pd.DataFrame,
    group_col: str,
    label_map: dict[str, str] | None = LABEL_MAP,
    loc: str = "center right",
    bbox_to_anchor: tuple[float, float] = (1.02, 0.5),
) -> None:
    """
    Add one figure-level legend for a categorical group column.
    """
    group_values = sorted(df[group_col].dropna().unique())
    default_cycle = plt.rcParams["axes.prop_cycle"].by_key()["color"]

    handles = [
        Line2D(
            [0],
            [0],
            marker="o",
            linestyle="",
            color=default_cycle[i % len(default_cycle)],
            label=f"{get_label(group_col, label_map)}={value}",
        )
        for i, value in enumerate(group_values)
    ]

    fig.legend(
        handles=handles,
        loc=loc,
        bbox_to_anchor=bbox_to_anchor,
        frameon=True,
    )

def add_global_group_legend_below(
    fig: plt.Figure,
    df: pd.DataFrame,
    group_col: str,
    label_map: dict[str, str] | None = LABEL_MAP,
    ncol: int | None = None,
) -> None:

        handles=handles,
        loc="lower center",
        bbox_to_anchor=(0.5, -0.02),
        ncol=len(handles) if ncol is None else ncol,
        frameon=False,

def plot_mfe_matches_vs_extra_pairs(
    df: pd.DataFrame,
    ax: plt.Axes,
    color_col: str = "E_att",
    run_col: str = "run_id",
    x_col: str = "mfe_matches",
    y_col: str = "mfe_extra_pairs",
    style: PlotStyle | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
    show_legend: bool = False,
    add_guides: bool = True,
    annotate_group_means: bool = True,
    mean_marker: str = "X",
    mean_marker_size: float = 180,
) -> plt.Axes:
    """
    Plot replicate-level mean MFE matches vs extra pairs.

    Small dots = one replicate (mean over rows passed to this function).
    Large X marker = mean across replicates for each color group.
    Optional text annotation shows the group mean coordinates.
    """
    if style is None:
        style = PlotStyle()

    required_cols = [run_col, x_col, y_col, color_col]
    missing_cols = [col for col in required_cols if col not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    # replicate-level means
    rep_df = (
        df.dropna(subset=required_cols)
        .groupby([run_col, color_col], as_index=False)
        .agg(
            x_mean=(x_col, "mean"),
            y_mean=(y_col, "mean"),
        )
    )

    default_cycle = plt.rcParams["axes.prop_cycle"].by_key()["color"]
    color_values = sorted(rep_df[color_col].dropna().unique())

    for idx, value in enumerate(color_values):
        d = rep_df[rep_df[color_col] == value]
        color = default_cycle[idx % len(default_cycle)]

        # small points = replicate means
        ax.scatter(
            d["x_mean"],
            d["y_mean"],
            s=80,
            alpha=0.75,
            edgecolors="black",
            linewidths=0.5,
            color=color,
            label=f"{get_label(color_col, label_map)}={value}",
        )

        # big point = mean across replicates
        mx = d["x_mean"].mean()
        my = d["y_mean"].mean()

        ax.scatter(
            mx,
            my,
            s=mean_marker_size,
            marker=mean_marker,
            color=color,
            edgecolors="black",
            linewidths=1.0,
            zorder=5,
        )

        if annotate_group_means:
            ax.annotate(
                f"({mx:.1f}, {my:.1f})",
                xy=(mx, my),
                xytext=(6, 6),
                textcoords="offset points",
                fontsize=9,
                ha="left",
                va="bottom",
            )

    if add_guides:
        ax.axhline(0, linestyle="--", linewidth=1, alpha=0.5)

    # integer ticks, denser axes
    x_max = rep_df["x_mean"].max()
    y_max = rep_df["y_mean"].max()

    ax.set_xlim(-0.2, np.ceil(x_max) + 0.5)
    ax.set_ylim(-0.2, np.ceil(y_max) + 0.5)

    ax.xaxis.set_major_locator(MultipleLocator(1))
    ax.yaxis.set_major_locator(MultipleLocator(1))

    ax.set_xlabel(get_label(x_col, label_map))
    ax.set_ylabel(get_label(y_col, label_map))

    if style.show_grid:
        ax.grid(True, alpha=style.grid_alpha)

    if show_legend:
        ax.legend(loc=style.legend_loc)

    return ax

def plot_parameter_metric_heatmap(
    df: pd.DataFrame,
    ax: plt.Axes,
    x_col: str = "E_att",
    y_col: str = "angle_k_ppp",
    value_col: str = "mfe_bp_distance",
    run_col: str = "run_id",
    style: PlotStyle | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
    cmap: str = "viridis_r",
    annotate: bool = True,
) -> plt.Axes:
    """
    Plot parameter heatmap from replicate-level final-window means.

    The input df should already be filtered to the desired final frame window.
    Each replicate is averaged first, then parameter-combination means are plotted.
    """
    if style is None:
        style = PlotStyle()

    required_cols = [x_col, y_col, value_col, run_col]
    missing_cols = [col for col in required_cols if col not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    rep_df = (
        df.dropna(subset=required_cols)
        .groupby([x_col, y_col, run_col], as_index=False)
        .agg(value_mean=(value_col, "mean"))
    )

    heat_df = (
        rep_df
        .groupby([y_col, x_col], as_index=False)

    ])

    ax.set_xlabel(get_label(x_col, label_map))
    ax.set_ylabel(get_label(y_col, label_map))

    if annotate:
        for row_idx, y_val in enumerate(pivot.index):
            for col_idx, x_val in enumerate(pivot.columns):
                value = pivot.loc[y_val, x_val]
                if pd.notna(value):
                    ax.text(
                        col_idx,
                        row_idx,
                        f"{value:.1f}",
                        ha="center",
                        va="center",
                        fontsize=style.tick_size,
                    )

    cbar = ax.figure.colorbar(im, ax=ax)
    cbar.set_label(get_label(value_col, label_map))

    return ax

def plot_mfe_distance_vs_rg(
    df: pd.DataFrame,
    ax: plt.Axes,
    color_col: str = "E_att",
    run_col: str = "run_id",
    x_col: str = "mfe_bp_distance",
    y_col: str = "rg",
    reference_rg: float | None = None,
    style: PlotStyle | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
    show_legend: bool = False,
    annotate_group_means: bool = True,
) -> plt.Axes:
    """
    Plot replicate-level mean MFE base-pair distance against Rg.

    x-axis: MFE base-pair distance, lower is better.
    y-axis: radius of gyration.
    Small dots: replicate means.
    Large X: mean across replicates for each color group.
    Horizontal dashed line: optional reference Rg.
    """
    if style is None:
        style = PlotStyle()

    required_cols = [run_col, color_col, x_col, y_col]
    missing_cols = [col for col in required_cols if col not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    rep_df = (
        df.dropna(subset=required_cols)
        .groupby([run_col, color_col], as_index=False)
        .agg(
            x_mean=(x_col, "mean"),
            y_mean=(y_col, "mean"),
        )
    )

    default_cycle = plt.rcParams["axes.prop_cycle"].by_key()["color"]
    color_values = sorted(rep_df[color_col].dropna().unique())

    for idx, value in enumerate(color_values):
        d = rep_df[rep_df[color_col] == value]
        color = default_cycle[idx % len(default_cycle)]

        ax.scatter(
            d["x_mean"],
            d["y_mean"],
            s=80,
            alpha=0.75,
            edgecolors="black",
            linewidths=0.5,
            color=color,
            label=f"{get_label(color_col, label_map)}={value}",
        )

        mx = d["x_mean"].mean()
        my = d["y_mean"].mean()

        ax.scatter(
            mx,
            my,
            s=180,
            marker="X",
            color=color,
            edgecolors="black",
            linewidths=1.0,
            zorder=5,
        )

        if annotate_group_means:
            ax.annotate(
                f"({mx:.1f}, {my:.1f})",
                xy=(mx, my),
                xytext=(6, 6),
                textcoords="offset points",
                fontsize=9,
                ha="left",
                va="bottom",
            )

    if reference_rg is not None:
        ax.axhline(
            reference_rg,
            linestyle="--",
            linewidth=1.5,
            alpha=0.8,
        )
        ax.text(
            0.98,
            reference_rg,
            f"ref Rg = {reference_rg:.2f}",
            transform=ax.get_yaxis_transform(),
            ha="right",
            va="bottom",
            fontsize=9,
        )

    ax.set_xlabel(get_label(x_col, label_map))
    ax.set_ylabel(get_label(y_col, label_map))

    if style.show_grid:
        ax.grid(True, alpha=style.grid_alpha)

    if show_legend:
        ax.legend(loc=style.legend_loc)

    return ax

def make_final_rg_replicate_summary(
    df: pd.DataFrame,
    fraction: float = 0.2,
    fixed: dict | None = None,
    run_col: str = "run_id",
    sort_col: str = "step",
    rg_col: str = "rg",
) -> pd.DataFrame:
    """
    Build one-row-per-replicate Rg summaries from the final fraction of frames.
    """
    d = df_filter(df, fixed)
    d = final_fraction_df(
        d,
        fraction=fraction,
        run_col=run_col,
        sort_col=sort_col,
    )

        "angle_k_ppp",

        "density",
        run_col,
    ]

    summary = (
        d.dropna(subset=[rg_col])
        .groupby(group_cols, as_index=False)
        .agg(
            rg_mean=(rg_col, "mean"),
            rg_median=(rg_col, "median"),

            rg_q90=(rg_col, lambda x: x.quantile(0.90)),
            n_frames=(rg_col, "size"),
        )
    )

    return summary

def plot_rg_replicate_summary(
    df: pd.DataFrame,
    ax: plt.Axes,
    x_col: str = "E_att",
    y_col: str = "rg_mean",
    y_low_col: str = "rg_q10",
    y_high_col: str = "rg_q90",
    reference_rg: float | None = None,
    style: PlotStyle | None = None,
    label_map: dict[str, str] | None = LABEL_MAP,
    annotate_means: bool = True,
    show_reference_label: bool = True,
) -> plt.Axes:
    """
    Plot replicate-level Rg summaries.

    Points are replicate mean Rg values.
    Thin vertical bars show within-replicate final-window 10–90% Rg range.
    Black horizontal ticks show group means across replicates.
    """
    if style is None:
        style = PlotStyle()

    required_cols = [x_col, y_col, y_low_col, y_high_col]
    missing_cols = [col for col in required_cols if col not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing columns: {missing_cols}")

    d = df.dropna(subset=required_cols).copy()

    x_values = sorted(d[x_col].dropna().unique())
    x_map = {value: i for i, value in enumerate(x_values)}

    rng = np.random.default_rng(42)

    for x_value in x_values:
        sub = d[d[x_col] == x_value]
        x_base = x_map[x_value]

        x_jit = x_base + rng.normal(0, style.jitter, size=len(sub))

        ax.vlines(
            x_jit,
            sub[y_low_col],
            sub[y_high_col],
            alpha=0.30,
            linewidth=1.0,
        )

        ax.scatter(
            x_jit,
            sub[y_col],
            s=70,
            alpha=0.85,
            edgecolors="black",
            linewidths=0.5,
        )

        group_mean = sub[y_col].mean()

        ax.hlines(
            group_mean,
            x_base - 0.25,
            x_base + 0.25,
            linewidth=2.5,
            color="black",
        )

        if annotate_means:
            ax.text(
                x_base + 0.28,
                group_mean,
                f"{group_mean:.2f}",
                ha="left",
                va="center",
                fontsize=9,
                color="black",
            )

    if reference_rg is not None:
        ax.axhline(
            reference_rg,
            linestyle="--",
            linewidth=1.5,
            alpha=0.8,
        )

        if show_reference_label:
            ax.text(
                0.98,
                0.96,
                f"ref Rg = {reference_rg:.2f}",
                transform=ax.transAxes,
                ha="right",
                va="top",
                fontsize=9,
                bbox={
                    "facecolor": "white",
                    "edgecolor": "none",
                    "alpha": 0.8,
                    "pad": 2,
                },
            )

    ax.set_xticks(range(len(x_values)))
    ax.set_xticklabels([
        f"{x:g}" if isinstance(x, (int, float, np.integer, np.floating)) else str(x)
        for x in x_values
    ])

    ax.set_xlabel(get_label(x_col, label_map))
    ax.set_ylabel("Radius of gyration")

    if style.show_grid:
        ax.grid(True, axis="y", alpha=style.grid_alpha)

    return ax
